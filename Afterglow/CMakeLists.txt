cmake_minimum_required(VERSION 3.20)
project(Afterglow VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Afterglow requires Windows x64. Select the x64 toolchain.")
endif()
add_library(Afterglow MODULE Source/Plugin.cpp Source/Editor.cpp)
target_include_directories(Afterglow PRIVATE third_party Resources)
set(bundle "${CMAKE_BINARY_DIR}/Afterglow.vst3")
if(WIN32)
 enable_language(RC)
 target_sources(Afterglow PRIVATE Resources/Afterglow.rc)
 target_compile_definitions(Afterglow PRIVATE UNICODE _UNICODE NOMINMAX)
 target_link_libraries(Afterglow PRIVATE user32 gdi32 gdiplus ole32)
 set(binary_dir "${bundle}/Contents/x86_64-win")
 set_target_properties(Afterglow PROPERTIES SUFFIX ".vst3")
elseif(UNIX AND NOT APPLE)
 set(binary_dir "${bundle}/Contents/x86_64-linux")
 target_compile_options(Afterglow PRIVATE -fvisibility=hidden)
else()
 message(FATAL_ERROR "Windows x64 is the release target. Linux is for headless tests.")
endif()
set_target_properties(Afterglow PROPERTIES PREFIX "" OUTPUT_NAME "Afterglow")
foreach(kind LIBRARY RUNTIME ARCHIVE)
 set_target_properties(Afterglow PROPERTIES ${kind}_OUTPUT_DIRECTORY "${binary_dir}")
 foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(Afterglow PROPERTIES ${kind}_OUTPUT_DIRECTORY_${config} "${binary_dir}")
 endforeach()
endforeach()
if(MSVC)
 target_compile_options(Afterglow PRIVATE /W4 /utf-8 /Zc:__cplusplus)
elseif(MINGW)
 target_compile_options(Afterglow PRIVATE -O3)
 target_link_options(Afterglow PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp -Wl,--dynamicbase -Wl,--nxcompat)
endif()
file(MAKE_DIRECTORY "${bundle}/Contents/Resources")
configure_file(Resources/moduleinfo.json "${bundle}/Contents/Resources/moduleinfo.json" COPYONLY)
configure_file(third_party/pluginterfaces/LICENSE.txt "${bundle}/Contents/Resources/VST3-SDK-LICENSE.txt" COPYONLY)
option(AFTERGLOW_BUILD_TESTS "Build host tests and artwork renderer" ON)
if(AFTERGLOW_BUILD_TESTS)
 enable_testing()
 add_executable(AfterglowHostTests Tests/HostTests.cpp)
 target_include_directories(AfterglowHostTests PRIVATE third_party)
 if(WIN32)
  target_link_libraries(AfterglowHostTests PRIVATE user32 gdi32)
 else()
  target_link_libraries(AfterglowHostTests PRIVATE dl)
 endif()
 if(MINGW)
  target_link_options(AfterglowHostTests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 if(MSVC)
  target_compile_options(AfterglowHostTests PRIVATE /utf-8 /Zc:__cplusplus)
 endif()
 add_dependencies(AfterglowHostTests Afterglow)
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME native_vst3_contract COMMAND AfterglowHostTests "$<TARGET_FILE:Afterglow>")
 endif()
 add_executable(AfterglowPreview Tests/RenderPreview.cpp)
 add_executable(AfterglowRealtimeTests Tests/RealtimeTests.cpp)
 add_executable(AfterglowEditorTests Tests/EditorTests.cpp)
 if(MINGW)
  target_link_options(AfterglowRealtimeTests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(AfterglowEditorTests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME realtime_allocations COMMAND AfterglowRealtimeTests)
  add_test(NAME editor_interaction COMMAND AfterglowEditorTests)
 endif()
endif()
install(DIRECTORY "${bundle}" DESTINATION .)
