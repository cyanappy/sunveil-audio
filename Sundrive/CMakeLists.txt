cmake_minimum_required(VERSION 3.20)
project(Sundrive VERSION 1.3.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Sundrive requires a 64-bit build. Select the x64 toolchain.")
endif()

add_library(Sundrive MODULE Source/Plugin.cpp Source/Editor.cpp)
target_include_directories(Sundrive PRIVATE third_party)
set_target_properties(Sundrive PROPERTIES PREFIX "" OUTPUT_NAME "Sundrive")

set(bundle "${CMAKE_BINARY_DIR}/Sundrive.vst3")
if(WIN32)
    set(binary_dir "${bundle}/Contents/x86_64-win")
    set_target_properties(Sundrive PROPERTIES SUFFIX ".vst3")
    target_compile_definitions(Sundrive PRIVATE UNICODE _UNICODE)
    target_link_libraries(Sundrive PRIVATE user32 gdi32)
elseif(UNIX AND NOT APPLE)
    # Internal DSP/interface validation build. The custom editor targets Windows.
    set(binary_dir "${bundle}/Contents/x86_64-linux")
    target_compile_options(Sundrive PRIVATE -fvisibility=hidden)
else()
    message(FATAL_ERROR "This release targets Windows. Linux is supported for headless tests only.")
endif()

foreach(kind LIBRARY RUNTIME ARCHIVE)
    set_target_properties(Sundrive PROPERTIES ${kind}_OUTPUT_DIRECTORY "${binary_dir}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(Sundrive PROPERTIES ${kind}_OUTPUT_DIRECTORY_${config} "${binary_dir}")
    endforeach()
endforeach()

if(MSVC)
    target_compile_options(Sundrive PRIVATE /W4 /utf-8 /Zc:__cplusplus)
elseif(MINGW)
    target_compile_options(Sundrive PRIVATE -O3 -fno-exceptions -fno-rtti)
    target_link_options(Sundrive PRIVATE -static -static-libgcc -static-libstdc++
        -Wl,--no-insert-timestamp -Wl,--dynamicbase -Wl,--nxcompat)
endif()

file(MAKE_DIRECTORY "${bundle}/Contents/Resources")
configure_file(Resources/moduleinfo.json "${bundle}/Contents/Resources/moduleinfo.json" COPYONLY)
configure_file(third_party/pluginterfaces/LICENSE.txt
    "${bundle}/Contents/Resources/VST3-SDK-LICENSE.txt" COPYONLY)

option(SUNDRIVE_BUILD_TESTS "Build native VST3 host tests and artwork renderer" ON)
if(SUNDRIVE_BUILD_TESTS)
    enable_testing()
    add_executable(SundriveHostTests Tests/HostTests.cpp)
    target_include_directories(SundriveHostTests PRIVATE third_party)
    if(WIN32)
        target_link_libraries(SundriveHostTests PRIVATE user32 gdi32)
    else()
        target_link_libraries(SundriveHostTests PRIVATE dl)
    endif()
    if(MINGW)
        target_link_options(SundriveHostTests PRIVATE -static -static-libgcc -static-libstdc++)
    elseif(MSVC)
        target_compile_options(SundriveHostTests PRIVATE /utf-8 /Zc:__cplusplus)
    endif()
    add_dependencies(SundriveHostTests Sundrive)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME vst3_host_tests COMMAND SundriveHostTests "$<TARGET_FILE:Sundrive>")
    endif()
    add_executable(SundrivePreview Tests/RenderPreview.cpp)
    if(MSVC)
        target_compile_options(SundrivePreview PRIVATE /utf-8)
    endif()
endif()

install(DIRECTORY "${bundle}" DESTINATION .)
