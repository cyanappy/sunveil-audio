cmake_minimum_required(VERSION 3.20)
project(Glassgarden VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Glassgarden requires Windows x64. Select the x64 toolchain.")
endif()
add_library(Glassgarden MODULE Source/Plugin.cpp Source/Editor.cpp)
target_include_directories(Glassgarden PRIVATE third_party Resources)
set(bundle "${CMAKE_BINARY_DIR}/Glassgarden.vst3")
if(WIN32)
 enable_language(RC)
 target_sources(Glassgarden PRIVATE Resources/Glassgarden.rc)
 target_compile_definitions(Glassgarden PRIVATE UNICODE _UNICODE NOMINMAX)
 target_link_libraries(Glassgarden PRIVATE user32 gdi32 gdiplus ole32)
 set(binary_dir "${bundle}/Contents/x86_64-win")
 set_target_properties(Glassgarden PROPERTIES SUFFIX ".vst3")
elseif(UNIX AND NOT APPLE)
 set(binary_dir "${bundle}/Contents/x86_64-linux")
 target_compile_options(Glassgarden PRIVATE -fvisibility=hidden)
else()
 message(FATAL_ERROR "Windows x64 is the release target. Linux is for headless tests.")
endif()
set_target_properties(Glassgarden PROPERTIES PREFIX "" OUTPUT_NAME "Glassgarden")
foreach(kind LIBRARY RUNTIME ARCHIVE)
 set_target_properties(Glassgarden PROPERTIES ${kind}_OUTPUT_DIRECTORY "${binary_dir}")
 foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(Glassgarden PROPERTIES ${kind}_OUTPUT_DIRECTORY_${config} "${binary_dir}")
 endforeach()
endforeach()
if(MSVC)
 target_compile_options(Glassgarden PRIVATE /W4 /utf-8 /Zc:__cplusplus)
elseif(MINGW)
 target_compile_options(Glassgarden PRIVATE -O3)
 target_link_options(Glassgarden PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp -Wl,--dynamicbase -Wl,--nxcompat)
endif()
file(MAKE_DIRECTORY "${bundle}/Contents/Resources")
configure_file(Resources/moduleinfo.json "${bundle}/Contents/Resources/moduleinfo.json" COPYONLY)
configure_file(third_party/pluginterfaces/LICENSE.txt "${bundle}/Contents/Resources/VST3-SDK-LICENSE.txt" COPYONLY)
option(GLASSGARDEN_BUILD_TESTS "Build host tests and artwork renderer" ON)
if(GLASSGARDEN_BUILD_TESTS)
 enable_testing()
 add_executable(GlassgardenHostTests Tests/HostTests.cpp)
 target_include_directories(GlassgardenHostTests PRIVATE third_party)
 if(WIN32)
  target_link_libraries(GlassgardenHostTests PRIVATE user32 gdi32)
 else()
  target_link_libraries(GlassgardenHostTests PRIVATE dl)
 endif()
 if(MINGW)
  target_link_options(GlassgardenHostTests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 if(MSVC)
  target_compile_options(GlassgardenHostTests PRIVATE /utf-8 /Zc:__cplusplus)
 endif()
 add_dependencies(GlassgardenHostTests Glassgarden)
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME native_vst3_contract COMMAND GlassgardenHostTests "$<TARGET_FILE:Glassgarden>")
 endif()
 add_executable(GlassgardenPreview Tests/RenderPreview.cpp)
 add_executable(GlassgardenRealtimeTests Tests/RealtimeTests.cpp)
 add_executable(GlassgardenEditorTests Tests/EditorTests.cpp)
 if(MINGW)
  target_link_options(GlassgardenRealtimeTests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(GlassgardenEditorTests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME realtime_allocations COMMAND GlassgardenRealtimeTests)
  add_test(NAME editor_interaction COMMAND GlassgardenEditorTests)
 endif()
endif()
install(DIRECTORY "${bundle}" DESTINATION .)
